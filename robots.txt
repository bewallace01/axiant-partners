# Every group below repeats the Disallow list on purpose.
# Under the robots.txt spec a crawler obeys ONLY its most specific matching
# group and ignores 'User-agent: *' entirely - so a group containing just
# 'Allow: /' grants that crawler access to the private directories.
#
# Access control is enforced in _redirects (these paths return 404).
# robots.txt is a request; _redirects is the actual block.

User-agent: *
Allow: /
Disallow: /tools/article_supplements/
Disallow: /tools/includes/
Disallow: /_outreach/
Disallow: /_marketing/
Disallow: /_analysis/
Disallow: /_components/
